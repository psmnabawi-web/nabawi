# ===================================================================
# KONTROL ENERGI STORE v1.1.5 - SEKALI COPY-PASTE DI GOOGLE CLOUD SHELL
#
# Project sudah dikunci ke: electric-control-bba
# Blok ini tidak menanyakan apa pun. Cukup paste lalu tunggu selesai.
#
# Perbaikan v1.1.5:
#   - Firestore rules dirombak agar muat di batas 1.000 ekspresi. Pada v1.1.3
#     SELURUH penyimpanan pembacaan dan pembelian token ditolak server.
#   - Pembagian bilangan bulat pada rules diperbaiki (kWh per 24 jam).
#   - Firebase Hosting API kini ikut diaktifkan.
#   - Anonymous Authentication diaktifkan otomatis, tidak lagi manual di konsol.
#   - Aplikasi dapat dipasang di layar HP (PWA) dengan ikon dan offline shell.
#
# Opsional, jalankan SEBELUM blok ini bila ingin App Check aktif:
#   export APPCHECK_SITE_KEY="site-key-reCAPTCHA-v3-anda"
#
# RESET DATA (opsional, menghapus seluruh pembacaan dan pembelian token):
#   export KES_RESET_DATA=1
#   export KES_RESET_CONFIRM=RESET
# Seluruh isi enam koleksi dicadangkan dulu ke berkas NDJSON di folder kerja
# sebelum dikosongkan. Tanpa kedua variabel itu, data existing tidak disentuh.
#
# Kegagalan deploy tidak akan menutup terminal utama.
# ===================================================================

bash <<'DEPLOY_INNER' 2>&1 | tee kontrol-energi-deploy.log
set -Eeuo pipefail

APP_RELEASE="1.1.5"
PROJECT_ID="${KES_PROJECT_ID:-electric-control-bba}"
APPCHECK_SITE_KEY="${APPCHECK_SITE_KEY:-}"
APPCHECK_PROVIDER="${APPCHECK_PROVIDER:-recaptcha-v3}"

STEP="Persiapan"
trap 'CODE=$?; echo; echo "================================================="; echo "DEPLOY GAGAL"; echo "Tahap : $STEP"; echo "Baris : $LINENO"; echo "Kode  : $CODE"; echo "Log   : kontrol-energi-deploy.log"; echo "================================================="; exit "$CODE"' ERR

echo "================================================="
echo " DEPLOY KONTROL ENERGI STORE v$APP_RELEASE"
echo " Project: $PROJECT_ID"
echo "================================================="

STEP="Mencari paket aplikasi"
echo "[1/13] Mencari paket v$APP_RELEASE..."
APP_DIR=""
ZIP_FILE="$(find . -maxdepth 1 -type f -name "Kontrol_Energi_Store_Firebase_v${APP_RELEASE}.zip" -printf '%T@ %p\n' 2>/dev/null | sort -nr | head -1 | cut -d' ' -f2- || true)"
if [ -n "${ZIP_FILE:-}" ]; then
  echo "ZIP: $ZIP_FILE"
  APP_DIR="$(mktemp -d "$PWD/kontrol-energi-store-deploy-v${APP_RELEASE}.XXXXXX")"
  unzip -o "$ZIP_FILE" -d "$APP_DIR" >/dev/null
elif [ -f public/app.js ] && [ -f scripts/seed-master.sh ]; then
  echo "Menjalankan dari folder paket yang sudah diekstrak."
  APP_DIR="$PWD"
else
  echo "Paket v$APP_RELEASE tidak ditemukan."
  echo "Upload Kontrol_Energi_Store_Firebase_v${APP_RELEASE}.zip ke Cloud Shell, lalu jalankan blok ini lagi."
  exit 21
fi
cd "$APP_DIR"

STEP="Memeriksa kelengkapan paket"
echo "[2/13] Memeriksa kelengkapan dan konsistensi versi..."
for required in public/master-electric.json public/app.js public/index.html \
  public/manifest.webmanifest public/sw.js public/icons/icon-512.png \
  scripts/seed-master.sh scripts/deploy.sh scripts/reset-data.sh \
  scripts/lib-gcloud-auth.sh firestore.rules firestore.indexes.json; do
  if [ ! -f "$required" ]; then
    echo "Paket tidak lengkap, file hilang: $required"
    exit 27
  fi
done
if ! grep -q "^Version: ${APP_RELEASE}$" VERSION.txt \
  || ! grep -q "const APP_VERSION = \"${APP_RELEASE}\";" public/app.js \
  || ! grep -q "const APP_VERSION = \"${APP_RELEASE}\";" public/sw.js \
  || ! grep -q "app.js?v=${APP_RELEASE}" public/index.html; then
  echo "Nomor versi paket tidak konsisten. Gunakan ZIP v$APP_RELEASE yang terbaru."
  exit 27
fi
# Rules v1.1.5 wajib membawa perbaikan batas ekspresi dan pembagian desimal.
if ! grep -q 'function stateUpdateOk(meterId)' firestore.rules \
  || ! grep -q 'd.usageKwh \* 24.0 / d.intervalHours' firestore.rules \
  || ! grep -q 'match /tokenPurchases/' firestore.rules \
  || ! grep -q 'match /tokenStates/' firestore.rules \
  || ! grep -q 'Menyiapkan state token 9 meter prabayar' scripts/seed-master.sh; then
  echo "firestore.rules bukan versi v1.1.5 yang sudah diperbaiki. Paket tercampur."
  exit 27
fi
if [ -e public/ocr ]; then
  echo "Paket masih membawa aset OCR lama dan tidak aman untuk rilis manual-only."
  exit 29
fi

STEP="Menjalankan inti deploy"
echo "Melanjutkan ke proses deploy bersama (scripts/deploy.sh)..."
export KES_PROJECT_ID="$PROJECT_ID"
export APPCHECK_SITE_KEY APPCHECK_PROVIDER
export KES_RESET_DATA="${KES_RESET_DATA:-}"
export KES_RESET_CONFIRM="${KES_RESET_CONFIRM:-}"
bash scripts/deploy.sh

DEPLOY_INNER

DEPLOY_STATUS=${PIPESTATUS[0]}
echo
if [ "$DEPLOY_STATUS" -ne 0 ]; then
  echo "Terminal tetap terbuka. Kirim 30 baris terakhir dari kontrol-energi-deploy.log untuk diperbaiki."
  echo "Jika tahap sinkronisasi master sudah selesai, datanya tetap aman dan tidak ganda. Jalankan blok yang sama untuk melanjutkan deploy."
else
  echo "Selesai. Terminal aman untuk ditutup."
fi
